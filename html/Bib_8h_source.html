121#define BIT_SET(byte, pos) (byte |= (1 << pos))
122// - (1 << pos) décale le bit "1" vers la gauche de "pos" positions (ex. : pos = 2 donne 00000100)
123// - |= effectue un OU binaire avec le byte d'origine → cela place le bit à 1 sans toucher aux autres

126#define BIT_CLEAR(byte, pos) (byte &= ~(1 << pos))
127// - (1 << pos) crée un masque avec un 1 à la position souhaitée
128// - ~ inverse tous les bits (donc un 0 à "pos", 1 ailleurs)
129// - &= applique ce masque en conservant les bits sauf celui à "pos" qui est mis à 0
